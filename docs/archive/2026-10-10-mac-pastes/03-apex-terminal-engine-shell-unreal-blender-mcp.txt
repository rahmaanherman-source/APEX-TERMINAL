import React, { useState, useEffect, useReducer, useRef, useCallback } from 'react';

const initial = {
  truth: 'UNKNOWN',
  panels: { gabby: true, proj: true, eng: false, conn: true, audit: true, sys: true, foley: false, timeline: false },
  engines: { unreal: 'OFF', blender: 'OFF', flutter: 'OFF', pixel: 'OFF' },
  audit: [],
  progress: 0,
  cmd: '',
  mcpTools: [],
};

function reducer(s, a) {
  switch (a.type) {
    case 'LOG': return { ...s, audit: [`${new Date().toLocaleTimeString()} ${a.msg}`, ...s.audit].slice(0, 80) };
    case 'TRUTH': return { ...s, truth: a.v };
    case 'PANEL': return { ...s, panels: { ...s.panels, [a.id]: !s.panels[a.id] } };
    case 'ENGINE': return { ...s, engines: { ...s.engines, [a.id]: a.v } };
    case 'PROG': return { ...s, progress: a.v };
    case 'CMD': return { ...s, cmd: a.v };
    case 'MCP_TOOLS': return { ...s, mcpTools: a.tools };
    default: return s;
  }
}

export default function ApexTerminal() {
  const [s, d] = useReducer(reducer, initial);
  const unrealWs = useRef(null);
  const pixelWs = useRef(null);

  const log = useCallback(msg => d({ type: 'LOG', msg }), []);
  const setTruth = v => d({ type: 'TRUTH', v });

  // —— Unreal Remote Control WebSocket (default :30020)
  const connectUnrealWS = () => {
    if (unrealWs.current?.readyState === 1) return;
    log('Unreal WS connecting ws://127.0.0.1:30020 …');
    try {
      const ws = new WebSocket('ws://127.0.0.1:30020');
      unrealWs.current = ws;
      ws.onopen = () => {
        d({ type: 'ENGINE', id: 'unreal', v: 'LINKED' });
        setTruth('PARTIAL');
        log('Unreal Remote Control WS open');
        // Example: probe via HTTP-over-WS
        ws.send(JSON.stringify({
          MessageName: 'http',
          Parameters: { Url: '/remote/info', Verb: 'GET', Body: {} },
          Id: 1
        }));
      };
      ws.onmessage = e => {
        log('Unreal WS ← ' + String(e.data).slice(0, 120));
      };
      ws.onerror = () => { log('Unreal WS error'); d({ type: 'ENGINE', id: 'unreal', v: 'ERR' }); };
      ws.onclose = () => { d({ type: 'ENGINE', id: 'unreal', v: 'OFF' }); log('Unreal WS closed'); };
    } catch (e) { log('Unreal WS fail: ' + e.message); d({ type: 'ENGINE', id: 'unreal', v: 'OFF' }); }
  };

  // —— Pixel Streaming signaling stub (common :8888)
  const connectPixel = () => {
    log('Pixel Streaming signaling ws://127.0.0.1:8888 …');
    try {
      const ws = new WebSocket('ws://127.0.0.1:8888');
      pixelWs.current = ws;
      ws.onopen = () => { d({ type: 'ENGINE', id: 'pixel', v: 'SIGNAL' }); log('Pixel signaling open — attach WebRTC player next'); };
      ws.onerror = () => { d({ type: 'ENGINE', id: 'pixel', v: 'OFF' }); log('Pixel signaling unreachable'); };
      ws.onclose = () => d({ type: 'ENGINE', id: 'pixel', v: 'OFF' });
    } catch (e) { log('Pixel fail: ' + e.message); }
  };

  // —— MCP protocol stubs (tools/list + tools/call) — Blender/Unreal MCP servers
  const mcpList = async (endpoint = 'http://127.0.0.1:9876') => {
    log('MCP tools/list → ' + endpoint);
    // Real MCP is usually stdio JSON-RPC; HTTP bridge example:
    try {
      const res = await fetch(endpoint + '/mcp', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ jsonrpc: '2.0', id: 1, method: 'tools/list', params: {} })
      });
      if (!res.ok) throw new Error(res.status);
      const data = await res.json();
      const tools = data?.result?.tools || [];
      d({ type: 'MCP_TOOLS', tools });
      d({ type: 'ENGINE', id: 'blender', v: tools.length ? 'MCP' : 'OFF' });
      log(`MCP tools: ${tools.length}`);
      setTruth(tools.length ? 'PARTIAL' : 'UNKNOWN');
    } catch {
      log('MCP unreachable — start Blender/Unreal MCP server');
      d({ type: 'ENGINE', id: 'blender', v: 'OFF' });
    }
  };

  const mcpCall = async (name, args = {}) => {
    log(`MCP tools/call ${name}`);
    // Same JSON-RPC shape; wire to your running MCP bridge
  };

  const action = (a) => {
    log(a.toUpperCase() + ' requested');
    if (a === 'verify') {
      setTruth('VERIFYING');
      setTimeout(() => { setTruth('NOT VERIFIED'); log('No live evidence'); }, 700);
    }
  };

  useEffect(() => {
    log('APEX React shell online — connectors offline');
    const t = setInterval(() => {}, 1000);

      clearInterval(t);
      unrealWs.current?.close();
      pixelWs.current?.close();
    };
  }, []);

      </div>
      {s.panels[id] && <div className="panel-b">{children}</div>}
    </div>
  );

  const pill = s.truth === 'UNKNOWN' || s.truth === 'NOT VERIFIED' ? 's-unknown'
    : s.truth === 'PARTIAL' || s.truth === 'VERIFYING' ? 's-warn' : 's-ok';

  return (

        <input className="search" value={s.cmd} onChange={e => d({ type: 'CMD', v: e.target.value })}
          onKeyDown={e => e.key === 'Enter' && (log('CMD: ' + s.cmd), d({ type: 'CMD', v: '' }))}
          placeholder="GABBY, WHAT ARE WE BUILDING?" />
        <span className={`status-pill ${pill}`}>STATUS: {s.truth}</span>
      </header>

      <nav className="left-rail">{/* icons same as before */}</nav>

            <div className="viewport-img">Godspeed viewport · Unreal Pixel Stream / Blender preview</div>
            <div className="toolbar">
              {['SCULPT','MODEL','TEXTURE','RIG','ANIMATE','LIGHT','RENDER'].map(t => (
                <button key={t} onClick={() => log('Tool: '+t)}>{t}</button>
              ))}
            </div>
          </div>
          <div className="side">
            <Panel id="gabby" title="CONCIERGE: GABBY">
              <p style={{color:'#8b9bb4'}}>Systems initializing. Command ready.</p>
              <button className="btn" onClick={() => log('New project')}>NEW PROJECT</button>
            </Panel>
            <Panel id="proj" title="PROJECT: GODSPEED">
              STATUS: {s.truth}

              ))}
            </Panel>
            <Panel id="conn" title="CONNECTORS">
              <button className="btn" onClick={connectUnrealWS}>Unreal RC WebSocket :30020</button>

Programming history → what died → what gives Apex the edge
Beginning (1940s–1970s)

Machine code / Assembly only.
First high-level: Fortran (1957), Lisp (1958), COBOL (1959), ALGOL, BASIC, Pascal.
C (1972) became the systems foundation.
Paradigm: write exactly what the hardware does. No abstraction safety net.

Middle (1980s–2010s)

C++ (1983) dominated engines and games.
OOP peak: Java, C#, Objective-C, Smalltalk influence.
Scripting layer appears: Lua, Python, JavaScript for tools and gameplay.
Web explosion: JS + PHP + later TypeScript.
Still mostly: manual memory, heavy boilerplate, separate toolchains, human writes every line.

Last ~10 years (especially 2020–2026)
Dead or dying practices

Pure manual memory management everywhere (C-style without borrow checking).
Objective-C, classic Visual Basic, Delphi, most proprietary BASICs.
Heavy reactive frameworks for ordinary concurrency (replaced by virtual threads / simpler models).
“Write all plumbing by hand” (DTOs, mappers, glue).
Monolithic editor UIs with no remote/AI control surface.
Binary-only asset pipelines that AI cannot read or generate.
Assuming the human is the only author of code.

What still works but is no longer enough alone

C++ as engine core (Unreal still lives here).
Python for tools and Blender.
TypeScript/React for the control surface.
Lua for lightweight scripting.

Current leg-up stack (2024–2026)

Rust / Zig for new performance-critical or safety-critical pieces (memory safety, tiny binaries, WASM, agent-friendly generation).
TypeScript + React for the Apex shell (fast UI, real state, WebSocket).
Python only where it already owns the domain (Blender MCP, tooling).
MCP (Model Context Protocol) as the universal AI ↔ engine bridge (Unreal, Blender, others already have servers).
WebSocket + Pixel Streaming / Remote Control instead of closed desktop-only editors.
Honest, evidence-based status (no fake green) + collapsible real-time panels.
Agent-first design: code and data formats that LLMs can reliably generate, verify, and drive.

Direct advantage for Apex

Keep the visual shell you already have (React state, collapse, honest status).
Talk to Unreal via Remote Control WebSocket (:30020) and Pixel Streaming signaling.
Talk to Blender via MCP / socket (:9876).
Expose every major action as MCP tools so any AI agent can drive the terminal.
Use Rust/Zig only for new high-performance bridges or WASM modules if needed; do not rewrite the UI.
Treat the terminal as the single control plane that plugs into Unreal, Blender, Flutter, Replit, Lovable, etc.

Bottom line

Old way = human writes C++/Python/JS inside closed editors.

Apex way = human + AI agents drive a live, honest, multi-engine control surface over WebSocket + MCP. That is the current gap most teams still miss.
